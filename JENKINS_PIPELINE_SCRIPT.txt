pipeline {
    agent any
    
    environment {
        DOCKERHUB_USERNAME = 'sarvnoorkaur'
        BACKEND_IMAGE = "${DOCKERHUB_USERNAME}/collabsphere-backend"
        FRONTEND_IMAGE = "${DOCKERHUB_USERNAME}/collabsphere-frontend"
    }
    
    stages {
        stage('Clone Repository') {
            steps {
                echo '📥 Cloning repository...'
                git branch: 'main', url: 'https://github.com/Sarvnoor-kaur/Collab'
            }
        }
        
        stage('Build Docker Images') {
            steps {
                echo '🔨 Building Docker images...'
                sh '''
                    docker build -t ${BACKEND_IMAGE}:latest -f docker/backend.Dockerfile ./backend
                    docker build -t ${FRONTEND_IMAGE}:latest -f docker/frontend.Dockerfile ./client
                '''
            }
        }
        
        stage('Push to Docker Hub') {
            steps {
                echo '📤 Pushing images to Docker Hub...'
                withCredentials([usernamePassword(
                    credentialsId: 'dockerhub-credentials',
                    usernameVariable: 'USER',
                    passwordVariable: 'PASS'
                )]) {
                    sh '''
                        echo $PASS | docker login -u $USER --password-stdin
                        docker push ${BACKEND_IMAGE}:latest
                        docker push ${FRONTEND_IMAGE}:latest
                    '''
                }
            }
        }
        
        stage('Load Images to Minikube') {
            steps {
                echo '📦 Loading images to Minikube...'
                sh '''
                    minikube image load ${BACKEND_IMAGE}:latest
                    minikube image load ${FRONTEND_IMAGE}:latest
                '''
            }
        }
        
        stage('Deploy to Kubernetes') {
            steps {
                echo '☸️ Deploying to Kubernetes...'
                sh '''
                    # Replace DockerHub username in deployment.yaml
                    sed -i "s|DOCKERHUB_USERNAME|${DOCKERHUB_USERNAME}|g" kubernetes/deployment.yaml
                    
                    # Apply Kubernetes manifests
                    kubectl apply -f kubernetes/deployment.yaml
                    kubectl apply -f kubernetes/service.yaml
                    
                    # Wait for deployments to be ready
                    echo "⏳ Waiting for deployments to be ready..."
                    kubectl rollout status deployment/backend -n collabsphere --timeout=300s
                    kubectl rollout status deployment/frontend -n collabsphere --timeout=300s
                '''
            }
        }
        
        stage('Verify Deployment') {
            steps {
                echo '✅ Verifying deployment...'
                sh '''
                    echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
                    echo "📊 Pods Status:"
                    kubectl get pods -n collabsphere
                    echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
                    echo "🌐 Services:"
                    kubectl get svc -n collabsphere
                    echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
                    
                    # Get Minikube IP and NodePorts
                    MINIKUBE_IP=$(minikube ip)
                    FRONTEND_PORT=$(kubectl get svc frontend-service -n collabsphere -o jsonpath='{.spec.ports[0].nodePort}')
                    BACKEND_PORT=$(kubectl get svc backend-service -n collabsphere -o jsonpath='{.spec.ports[0].nodePort}')
                    
                    echo "🌐 Frontend: http://${MINIKUBE_IP}:${FRONTEND_PORT}"
                    echo "🔧 Backend: http://${MINIKUBE_IP}:${BACKEND_PORT}"
                    echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
                    
                    # Health check
                    sleep 30
                    curl -f http://${MINIKUBE_IP}:${BACKEND_PORT}/api/health || exit 1
                    echo "✅ Health check passed!"
                '''
            }
        }
        
        stage('Cleanup') {
            steps {
                echo '🧹 Cleaning up...'
                sh '''
                    docker system prune -f
                '''
            }
        }
    }
    
    post {
        success {
            echo '✅ Pipeline completed successfully!'
            echo '━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━'
            echo '☸️  CollabSphere deployed to Kubernetes!'
            echo '━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━'
            sh '''
                MINIKUBE_IP=$(minikube ip)
                FRONTEND_PORT=$(kubectl get svc frontend-service -n collabsphere -o jsonpath='{.spec.ports[0].nodePort}')
                BACKEND_PORT=$(kubectl get svc backend-service -n collabsphere -o jsonpath='{.spec.ports[0].nodePort}')
                echo "🌐 Frontend: http://${MINIKUBE_IP}:${FRONTEND_PORT}"
                echo "🔧 Backend: http://${MINIKUBE_IP}:${BACKEND_PORT}"
            '''
        }
        failure {
            echo '❌ Deployment failed! Check logs above.'
        }
        always {
            echo '🔓 Logging out from Docker Hub...'
            sh 'docker logout || true'
        }
    }
}
